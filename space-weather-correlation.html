// Calcolo effemeridi reali IMMEDIATO (senza attendere la rete) function calculatePlanetaryAlignments() { const now = new Date(); const d = (now - Date.parse("2000-01-01T12:00:00Z")) / 86400000; const merc = (252.25 + 4.092338 * d) % 360; const venus = (181.98 + 1.602130 * d) % 360; const mars = (355.45 + 0.524039 * d) % 360; const jup = (34.35 + 0.083085 * d) % 360; let alignmentStatus = "Dispersione Standard"; if (Math.abs(merc - venus) < 12 || Math.abs(venus - mars) < 12 || Math.abs(merc - jup) < 15) { alignmentStatus = "Congiunzione Planetaria Attiva"; } document.getElementById('planetary-alignment').innerText = alignmentStatus; document.getElementById('planetary-details').innerText = `Mer: ${Math.round(merc)}° | Ven: ${Math.round(venus)}° | Mar: ${Math.round(mars)}° | Gio: ${Math.round(jup)}°`; } // Fetch con gestione di fallback per le API Reali async function fetchRealData() { // Esegue subito il calcolo astronomico calculatePlanetaryAlignments(); // 1. Indice Kp NOAA Reale try { const kpRes = await fetch('https://services.swpc.noaa.gov/json/planetary_k_index_1h.json'); const kpData = await kpRes.json(); const latestKp = kpData[kpData.length - 1]; document.getElementById('kp-index-val').innerText = `Kp ${latestKp.kp}`; document.getElementById('kp-details').innerText = `NOAA UTC: ${latestKp.time_tag.replace('T', ' ').substring(11, 16)}`; } catch (e) { document.getElementById('kp-index-val').innerText = "Errore Rete NOAA"; } // 2. Vento Solare NOAA Reale try { const windRes = await fetch('https://services.swpc.noaa.gov/products/solar-wind/plasma-6-hour.json'); const windData = await windRes.json(); const latestWind = windData[windData.length - 1]; const speed = latestWind[1] ? Math.round(latestWind[1]) : 'N/D'; const density = latestWind[2] ? parseFloat(latestWind[2]).toFixed(2) : 'N/D'; document.getElementById('solar-wind-speed').innerText = `${speed} km/s`; document.getElementById('solar-wind-details').innerText = `Densità: ${density} p/cm³ (NOAA SWPC)`; } catch (e) { document.getElementById('solar-wind-speed').innerText = "Errore Rete NOAA"; } // 3. Terremoti USGS Reali (M > 4.5) try { const usgsRes = await fetch('https://earthquake.usgs.gov/earthquakes/feed/v1.0/summary/4.5_day.geojson'); const usgsData = await usgsRes.json(); document.getElementById('usgs-quake-count').innerText = `${usgsData.features.length} Eventi (M > 4.5)`; if (usgsData.features.length > 0) { const maxMag = Math.max(...usgsData.features.map(f => f.properties.mag)); document.getElementById('usgs-max-mag').innerText = `Max Magnitudo: M ${maxMag.toFixed(1)}`; } } catch (e) { document.getElementById('usgs-quake-count').innerText = "Errore Rete USGS"; } } // Avvio immediato al caricamento document.addEventListener("DOMContentLoaded", fetchRealData);
